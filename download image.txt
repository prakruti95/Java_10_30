# 🌐 3. ImageDownloadServlet.java

```java id="imgdl01"
package com.example.servlet;

import java.io.*;
import java.net.URL;
import java.sql.Connection;
import java.sql.PreparedStatement;

import javax.servlet.*;
import javax.servlet.http.*;

public class ImageDownloadServlet extends HttpServlet {

    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        String imageUrl = request.getParameter("url");
        String fileName = "img_" + System.currentTimeMillis() + ".jpg";

        String savePath = getServletContext().getRealPath("/images") 
                        + File.separator + fileName;

        try {
            // Download image
            URL url = new URL(imageUrl);
            InputStream in = url.openStream();
            FileOutputStream out = new FileOutputStream(savePath);

            byte[] buffer = new byte[4096];
            int bytesRead;

            while ((bytesRead = in.read(buffer)) != -1) {
                out.write(buffer, 0, bytesRead);
            }

            in.close();
            out.close();

            // Save path in DB
            Connection con = DBConnection.getConnection();
            PreparedStatement ps = con.prepareStatement(
                "INSERT INTO images (name, path) VALUES (?, ?)"
            );

            ps.setString(1, fileName);
            ps.setString(2, "images/" + fileName);

            ps.executeUpdate();

            response.sendRedirect("list");

        } catch (Exception e) {
            e.printStackTrace();
            response.getWriter().println("Error downloading image");
        }
    }
}
```

# 📋 4. ImageListServlet.java

```java id="imglist01"
package com.example.servlet;

import java.io.IOException;
import java.sql.*;
import java.util.*;

import javax.servlet.*;
import javax.servlet.http.*;

public class ImageListServlet extends HttpServlet {

    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        List<String> images = new ArrayList<>();

        try {
            Connection con = DBConnection.getConnection();
            Statement st = con.createStatement();

            ResultSet rs = st.executeQuery("SELECT * FROM images");

            while (rs.next()) {
                images.add(rs.getString("path"));
            }

        } catch (Exception e) {
            e.printStackTrace();
        }

        request.setAttribute("images", images);
        RequestDispatcher rd = request.getRequestDispatcher("display.jsp");
        rd.forward(request, response);
    }
}
```

# 🚀 How to Run

1. Import project into Eclipse
2. Add MySQL connector `.jar`
3. Update DB username/password
4. Create `images` folder inside WebContent
5. Run on Tomcat server
6. Open:

```
http://localhost:8080/ImageApp/index.jsp
```
